cmake_minimum_required(VERSION 3.8)
project(mid360_lio_adapter)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(livox_ros_driver2 REQUIRED)
add_executable(timing_adapter src/adapter.cpp)
target_compile_features(timing_adapter PUBLIC cxx_std_17)
ament_target_dependencies(timing_adapter rclcpp sensor_msgs livox_ros_driver2)
install(TARGETS timing_adapter DESTINATION lib/${PROJECT_NAME})
ament_package()
