cmake_minimum_required(VERSION 3.16)
project(rover_companion_bridge)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(livox_ros_driver2 REQUIRED)
add_executable(sensor_ingress src/sensor_ingress.cpp)
target_compile_features(sensor_ingress PUBLIC cxx_std_17)
ament_target_dependencies(sensor_ingress rclcpp sensor_msgs)
add_executable(simulation_adapter src/simulation_adapter.cpp)
target_compile_features(simulation_adapter PUBLIC cxx_std_17)
ament_target_dependencies(simulation_adapter rclcpp sensor_msgs livox_ros_driver2)
install(TARGETS sensor_ingress simulation_adapter DESTINATION lib/${PROJECT_NAME})
ament_package()
